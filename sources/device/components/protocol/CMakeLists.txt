include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/Protocol.cmake")
set(generated "${CMAKE_BINARY_DIR}/generated/protocol")
set(version_header "${CMAKE_BINARY_DIR}/generated/ek_build_version.h")
execute_process(COMMAND git -C "${PROJECT_ROOT}" rev-parse HEAD
 OUTPUT_VARIABLE fw_git_sha1 OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE fw_git_result)
if(DEFINED ENV{EK_GIT_SHA1} AND NOT "$ENV{EK_GIT_SHA1}" STREQUAL "")
 set(fw_git_sha1 "$ENV{EK_GIT_SHA1}")
 set(fw_git_result 0)
endif()
string(LENGTH "${fw_git_sha1}" fw_git_length)
if(NOT fw_git_result EQUAL 0 OR NOT fw_git_length EQUAL 40 OR NOT fw_git_sha1 MATCHES "^[0-9a-fA-F]+$")
 string(LENGTH "${EK_GIT_SHA1}" fw_fallback_length)
 if(DEFINED EK_GIT_SHA1 AND fw_fallback_length EQUAL 40 AND EK_GIT_SHA1 MATCHES "^[0-9a-fA-F]+$")
  set(fw_git_sha1 "${EK_GIT_SHA1}")
 else()
  message(FATAL_ERROR "Cannot determine full Git SHA-1 for device firmware version")
 endif()
endif()
string(TIMESTAMP fw_build_timestamp "%Y%m%dT%H%M%SZ" UTC)
string(SUBSTRING "${fw_git_sha1}" 0 7 fw_short_git_sha)
set(fw_version "${fw_short_git_sha}_${fw_build_timestamp}")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${version_header}" "#pragma once\n#define EK_FW_VERSION_STRING \"${fw_version}\"\n")
if(CMAKE_BUILD_EARLY_EXPANSION)
 idf_component_register(REQUIRES mbedtls)
 return()
endif()
file(MAKE_DIRECTORY "${generated}")
generate_c_protocol("${generated}")
set(nanopb "${PROJECT_ROOT}/build/device/deps/nanopb")
if(NOT EXISTS "${nanopb}/pb_decode.c")
 message(FATAL_ERROR "Run scripts/wsl/bootstrap.sh to prepare pinned nanopb runtime")
endif()
idf_component_register(SRCS src/frame.c src/message_validate.c src/ek_crypto.c "${generated}/ek.pb.c"
 "${nanopb}/pb_common.c" "${nanopb}/pb_encode.c" "${nanopb}/pb_decode.c"
 INCLUDE_DIRS include "${generated}" "${CMAKE_BINARY_DIR}/generated" "${nanopb}" REQUIRES mbedtls)
add_dependencies(${COMPONENT_LIB} proto-c)
