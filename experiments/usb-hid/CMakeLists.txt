cmake_minimum_required(VERSION 3.24)
set(IDF_TARGET esp32s3)
if(NOT DEFINED SDKCONFIG)
    get_filename_component(config_dir "${CMAKE_CURRENT_LIST_DIR}/../../build/device/config" ABSOLUTE)
    file(MAKE_DIRECTORY "${config_dir}")
    set(SDKCONFIG "${config_dir}/hid-selftest.sdkconfig" CACHE FILEPATH "HID test configuration")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../sources/device/components/hid_reports")
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" STREQUAL "6.0.3")
    message(FATAL_ERROR "HID bring-up requires ESP-IDF v6.0.3")
endif()
project(hid-selftest VERSION 0.1.0)
